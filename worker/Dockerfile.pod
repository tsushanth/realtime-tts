# RunPod Pod (always-on) variant of the worker — runs server.py (direct WS) instead of
# handler.py (Serverless job API), to avoid RunPod Serverless's ~7s per-request dispatch
# latency (see ../DECISIONS.md). Separate from ../Dockerfile because Pods need an SSH
# daemon RunPod's infra expects for status/runtime reporting — the plain python:3.11-slim
# base used for Serverless has none, which is why three straight Pod deploys never
# reported `runtime` at all (confirmed via RunPod's own docs: "most custom templates do
# not have SSH daemon installed and configured" — this wasn't a networking/WS bug).
FROM python:3.11-slim

RUN apt-get update && apt-get install -y --no-install-recommends curl espeak-ng openssh-server \
    && mkdir -p /var/run/sshd \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

# Three real bugs found and fixed here, in order, all confirmed live via SSH on a
# running pod before being baked in (see DECISIONS.md for the full debugging trail):
#
# 1. onnxruntime-gpu and plain onnxruntime install to the same import path
#    (site-packages/onnxruntime/) and are NOT safe to have coinstalled - kokoro-onnx pulls
#    in plain onnxruntime transitively, which silently wins the file collision and
#    disables CUDA. Fix: install onnxruntime-gpu FIRST, then kokoro-onnx's other real
#    deps (`pip show kokoro-onnx`: espeakng-loader, numpy, phonemizer - deliberately
#    omitting onnxruntime), then kokoro-onnx itself with --no-deps.
# 2. Unpinned `onnxruntime-gpu` resolved to 1.29.0, which requires CUDA 13.x - but
#    real CUDA-13 pip packages (nvidia-cublas-cu13 etc.) are still unpublished stubs
#    (literally version "0.0.1" on PyPI as of this build). Pinned to 1.20.2, the newest
#    version confirmed to target CUDA 12.x + cuDNN 9.x, which DOES have real packages.
# 3. `nvidia-cuda-nvrtc-cu12` (provides libnvrtc.so.12, needed for CUDA JIT) was missing
#    from the original package list entirely - onnxruntime's CUDA EP load error names
#    its missing .so files one at a time, so this took iterative fixing to fully surface.
COPY requirements.txt .
RUN pip install --no-cache-dir 'onnxruntime-gpu==1.20.2' espeakng-loader phonemizer && \
    pip install --no-cache-dir --no-deps kokoro-onnx==0.6.1 && \
    pip install --no-cache-dir soundfile numpy websockets runpod \
      nvidia-cuda-runtime-cu12 nvidia-cuda-nvrtc-cu12 nvidia-cudnn-cu12 nvidia-cublas-cu12 \
      nvidia-cufft-cu12 nvidia-curand-cu12 nvidia-cusparse-cu12 nvidia-cusolver-cu12

ENV LD_LIBRARY_PATH="/usr/local/lib/python3.11/site-packages/nvidia/cudnn/lib:/usr/local/lib/python3.11/site-packages/nvidia/cublas/lib:/usr/local/lib/python3.11/site-packages/nvidia/cuda_runtime/lib:/usr/local/lib/python3.11/site-packages/nvidia/cuda_nvrtc/lib:/usr/local/lib/python3.11/site-packages/nvidia/cufft/lib:/usr/local/lib/python3.11/site-packages/nvidia/curand/lib:/usr/local/lib/python3.11/site-packages/nvidia/cusparse/lib:/usr/local/lib/python3.11/site-packages/nvidia/cusolver/lib:${LD_LIBRARY_PATH}"

COPY synth.py server.py ./

RUN mkdir -p models && \
    curl -sL -o models/kokoro-v1.0.onnx \
      https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.0/kokoro-v1.0.onnx && \
    curl -sL -o models/voices-v1.0.bin \
      https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.0/voices-v1.0.bin

COPY start-pod.sh /start-pod.sh
RUN chmod +x /start-pod.sh

CMD ["/start-pod.sh"]
