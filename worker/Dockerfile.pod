# RunPod Pod (always-on) variant of the worker — runs server.py (direct WS) instead of
# handler.py (Serverless job API), to avoid RunPod Serverless's ~7s per-request dispatch
# latency (see ../DECISIONS.md). Separate from ../Dockerfile because Pods need an SSH
# daemon RunPod's infra expects for status/runtime reporting — the plain python:3.11-slim
# base used for Serverless has none, which is why three straight Pod deploys never
# reported `runtime` at all (confirmed via RunPod's own docs: "most custom templates do
# not have SSH daemon installed and configured" — this wasn't a networking/WS bug).
FROM python:3.11-slim

RUN apt-get update && apt-get install -y --no-install-recommends curl espeak-ng openssh-server \
    && mkdir -p /var/run/sshd \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt onnxruntime-gpu

ENV LD_LIBRARY_PATH="/usr/local/lib/python3.11/site-packages/nvidia/cudnn/lib:/usr/local/lib/python3.11/site-packages/nvidia/cublas/lib:/usr/local/lib/python3.11/site-packages/nvidia/cuda_runtime/lib:/usr/local/lib/python3.11/site-packages/nvidia/cufft/lib:/usr/local/lib/python3.11/site-packages/nvidia/curand/lib:/usr/local/lib/python3.11/site-packages/nvidia/cusparse/lib:/usr/local/lib/python3.11/site-packages/nvidia/cusolver/lib:${LD_LIBRARY_PATH}"

COPY synth.py server.py .

RUN mkdir -p models && \
    curl -sL -o models/kokoro-v1.0.onnx \
      https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.0/kokoro-v1.0.onnx && \
    curl -sL -o models/voices-v1.0.bin \
      https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.0/voices-v1.0.bin

COPY start-pod.sh /start-pod.sh
RUN chmod +x /start-pod.sh

CMD ["/start-pod.sh"]
